
So for this, the trick is basically to calculate the left products - product of elements to the left of each element i.
Then calculate the right product - product of elements to the left of each element i
There's a way to achieve it via creating  leftProduct and rightProduct arrays which is okayish but we pretty much want to do this in constant space (output arr doesnt count.

```js

const productExceptSelf = (nums) => { 
	
	const res = new Array(nums.length)
	res[0] = 1
	
	/*
	This is how we find the left product.
	We start at i = 1 because there are no elements to the left
	of index 0 so we initialize index 0 to 1
	*/
	for(let i = 1; i < nums.length; i++) {		
		res [i] = nums[i-1] * res[i-1]	
	}
	
	//do the backwards pass to calculate right products
	
	let rightProduct = 1
	
	for(let i = nums.length - 1; i>-1; i--) {		
		res[i] = nums[i] * rightProduct
		/*
			think of rightProduct as storing the cumulative
			product of elements to the right as of point i
		
		*/
		rightProduct = nums[i] * rightProduct		
	}
	
	return res
	


}
```
