Given two strings `s` and `t` of lengths `m` and `n` respectively, return _the **minimum window**_ **_substring_** _of_ `s` _such that every character in_ `t` _(**including duplicates**) is included in the window_. If there is no such substring, return _the empty string_ `""`.

```js
var minWindow = function(s, t) {
    // Step 1: Build a frequency map for all characters in target string t
    let map = {};
    for (let char of t) {
        map[char] = (map[char] || 0) + 1;
    }

    // Initialize pointers and tracking variables
    let left = 0, right = 0;
    let len = Infinity;
    
    // count tracks how many unique characters from t we still need to satisfy
    let count = Object.keys(map).length;
    let minWindowStr = "";

    // Step 2: Expand the window by moving the right pointer forward
    while (right < s.length) {
        let letter = s[right];
        
        // If the current character is part of our target map, decrement its count
        if (letter in map) {
            map[letter]--;
            // If we have fulfilled all required instances of this specific character, decrement count
            if (map[letter] === 0) {
                count--;
            }
        }
        right++;

        // Step 3: Once count reaches 0, we have a valid window containing all required characters
        while (count === 0) {
            // Update our minimum window if the current one is shorter than previous ones
            if (right - left < len) {
                len = right - left;
                minWindowStr = s.slice(left, right);
            }

            // Step 4: Shrink the window from the left to find the smallest possible valid substring
            let leftLetter = s[left];
            if (leftLetter in map) {
                map[leftLetter]++;
                // If a required character count goes above 0, we are missing a target character again
                if (map[leftLetter] > 0) {
                    count++;
                }
            }
            left++;
        }
    }

    return minWindowStr;
};

```

/**

* @param {string} s

* @param {string} t

* @return {string}

*/

var minWindow = function(s, t) {

const map = {}

let minString = ''

let minStringLen = Infinity

for(const char of t) {
	map[char] = (map[char] || 0) + 1
}

let count = Object.keys(map).length 

let start = 0
for(let end = 0; end < s.length; end++) {
	const char = s[end]
	map[char]--
	if(map[char] === 0) count--
  

	while(count === 0) {
	if((end - start+1) < minStringLen) {
		minString = s.slice(start,end+1)
		minStringLen = end - start + 1
	}

	map[s[start]]++
	if(map[s[start]] > 0) count++
	start++

}

}

return minString

};
